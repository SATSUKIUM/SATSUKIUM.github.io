 100 //______________________________________________________________________________
 101 bool FileSink::CompressMultipartData(FairMQParts &rawMsgParts, int index)
 102 {
 103     if (fStopRequested) {
 104         return false;
 105     }
 106     FairMQParts compressedMsgParts;
 107     if (fMergeMessage) {
 108         auto len = MessageUtil::TotalLength(rawMsgParts);
 109         auto buf = fWorker->NewMessage(len);
 110         auto bufFirst = reinterpret_cast<char *>(buf->GetData());
 111         auto bufLast = bufFirst + len;
 112         auto itr = bufFirst;
 113 
 114         for (const auto &m : rawMsgParts) {
 115             auto first = reinterpret_cast<char *>(m->GetData());
 116             auto last = first + m->GetSize();
 117             // std::for_each(first, last, HexDump());
 118             std::copy(first, last, itr);
 119             itr += m->GetSize();
 120         }
 121         if (bufLast != itr) {
 122             LOG(error) << __LINE__ << ":" << __func__ << " index = " << index << ": failed to copy buffer";
 123         }
 124         auto compressed = std::make_unique<std::vector<char>>(fCompress(bufFirst, len));
 125         fSize += len;
 126         fCompressedSize += compressed->size();
 127         compressedMsgParts.AddPart(MessageUtil::NewMessage(*fWorker, std::move(compressed)));
 128     } else {
 129         for (const auto &m : rawMsgParts) {
 130             fSize += m->GetSize();
 131             auto first = reinterpret_cast<char *>(m->GetData());
 132             auto compressed = std::make_unique<std::vector<char>>(fCompress(first, m->GetSize()));
 133             fCompressedSize += compressed->size();
 134             compressedMsgParts.AddPart(MessageUtil::NewMessage(*fWorker, std::move(compressed)));
 135         }
 136     }
 137 
 138     // LOG(debug) << __LINE__ << ":" << __func__ << " index = " << index << " worker output channels send";
 139     fWorker->fWorkerOutputChannels[index]->Send(compressedMsgParts, fMQTimeoutMS);
 140     // LOG(debug) << __LINE__ << ":" << __func__ << " index = " << index << " worker output channels send done";
 141     return true;

 455             fClient = std::make_shared<sw::redis::Redis>(registryUri);

 521     LOG(debug) << "FileSink::Header.magic            : " << fFileSinkHeader.magic;
 522     //LOG(debug) << "FileSink::Header.size             : " << fFileSinkHeader.size;
 523     LOG(debug) << "FileSink::Header.hLength          : " << fFileSinkHeader.hLength;
 524     LOG(debug) << "FileSink::Header.fairMQDeviceType : " << fFileSinkHeader.fairMQDeviceType;
 525     LOG(debug) << "FileSink::Header.runNumber        : " << fFileSinkHeader.runNumber;
 526     LOG(debug) << "FileSink::Header.startUnixtime    : " << fFileSinkHeader.startUnixtime;
 527     LOG(debug) << "FileSink::Header.stopUnixtime     : " << fFileSinkHeader.stopUnixtime;
 528     LOG(debug) << "FileSink::Header.comments         : " << fFileSinkHeader.comments;
 529     fFile->Write(reinterpret_cast<char *>(&fFileSinkHeader), sizeof(FileSinkHeader::Header));
 530 }
